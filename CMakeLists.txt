# ============================================================
# Heat-Simulation
# 二维热传导数值模拟
# ============================================================

# 指定 CMake 的最低版本要求
cmake_minimum_required(VERSION 3.20)

# 定义项目名称
# LANGUAGES CXX 表示这个项目使用 C++
project(HeatSimulation LANGUAGES CXX)


# ============================================================
# C++ 编译标准
# ============================================================

# 使用 C++17 标准
set(CMAKE_CXX_STANDARD 17)

# 要求必须支持 C++17
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 禁止 CMake 自动降级到更低版本的 C++ 标准
set(CMAKE_CXX_EXTENSIONS OFF)


# ============================================================
# 创建可执行程序
# ============================================================

# heat_sim 是最终生成的可执行文件名称
#
# src/main.cpp 是这个可执行程序的源代码
add_executable(heat_sim
    src/main.cpp
)
